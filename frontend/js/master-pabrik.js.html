function renderMasterPabrik( data ){ const container = $("masterPabrikTable"); if(!container){ return; } if(!data.length){ container.innerHTML = `
Belum ada pabrik
`; return; } container.innerHTML = ` ${data.map( (item,index) => ` ` ).join("")}
No Nama Pabrik Alamat Kontak Keterangan
${index+1} ${escapeHtml( item.namaPabrik || "-" )} ${escapeHtml( item.alamat || "-" )} ${escapeHtml( item.kontak || "-" )} ${escapeHtml( item.keterangan || "-" )}
`; } function saveMasterPabrikForm(){ const data = { namaPabrik: $("masterPabrikNama") .value .trim(), alamat: $("masterPabrikAlamat") .value .trim(), kontak: $("masterPabrikKontak") .value .trim(), keterangan: $("masterPabrikKeterangan") .value .trim() }; if(!data.namaPabrik){ showToast( "Nama pabrik wajib diisi.", "error" ); return; } showLoading( "Menyimpan pabrik..." ); callServer( "saveMasterPabrik", [data], function(result){ hideLoading(); if( !result || result.success === false ){ showToast( result?.message || "Gagal menyimpan pabrik.", "error" ); return; } showToast( "Pabrik berhasil disimpan." ); $("masterPabrikNama").value = ""; $("masterPabrikAlamat").value = ""; $("masterPabrikKontak").value = ""; $("masterPabrikKeterangan").value = ""; loadMasterPabrik(); }, function(error){ hideLoading(); showToast( error?.message || "Gagal menyimpan pabrik.", "error" ); } ); } /* ========================================================= PEMBUKUAN ========================================================= */
